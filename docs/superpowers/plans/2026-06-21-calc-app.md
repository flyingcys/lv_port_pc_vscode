# 计算器 App 复刻实现计划 (calc.html → LVGL v9 + 桌面接入)

> **For agentic workers:** REQUIRED SUB-SKILL: 用 superpowers:subagent-driven-development（推荐）或 superpowers:executing-plans 逐任务实现。步骤用 checkbox（`- [ ]`）跟踪。

**Goal:** 把 `design-ui/calc/calc.html` 复刻为 LVGL v9 原生计算器，接入桌面，点 "计算器" 图标打开，三档分辨率自适应。

**Tech Stack:** LVGL v9, C99, `<math.h>`, lv_font_conv, headless Chrome 截图比对。

- [ ] **Step 4: 写 calc_engine.c 实现**

```c
#include "calc_engine.h"
#include <math.h>
#include <stdlib.h>
#include <string.h>
#include <stdio.h>

static void set_current(calc_engine_t * e, const char * s) {
    snprintf(e->current, sizeof(e->current), "%s", s);
}

static void update_expression_op(calc_engine_t * e) {
    if(e->op == 0 || e->previous[0] == '\0') {
        e->expression[0] = '\0';
        return;
    }
    snprintf(e->expression, sizeof(e->expression), "%s %c", e->previous, e->op);
}

static double compute(double prev, double curr, char op) {
    switch(op) {
        case '+': return prev + curr;
        case '-': return prev - curr;
        case '×': return prev * curr;
        case '÷': return prev / curr;
        case '^': return pow(prev, curr);
        default:  return curr;
    }
}

static void finish_result(calc_engine_t * e, double result) {
    if(isnan(result) || !isfinite(result)) {
        set_current(e, "Error");
    } else {
        /* %.12g 近似 toPrecision(12)，%g 自动去尾零 */
        snprintf(e->current, sizeof(e->current), "%.12g", result);
        /* 模拟 JS parseFloat(result.toPrecision(12))：去掉无意义，%g 已处理 */
    }
    e->op = 0;
    e->previous[0] = '\0';
    e->should_reset = true;
    e->is_result = true;
    update_expression_op(e);
}

static bool is_error(const calc_engine_t * e) {
    return strcmp(e->current, "Error") == 0;
}

void calc_engine_init(calc_engine_t * e) {
    memset(e, 0, sizeof(*e));
    set_current(e, "0");
}

static void handle_number(calc_engine_t * e, char ch) {
    if(is_error(e)) {
        set_current(e, "0");
    }
    if(e->should_reset || strcmp(e->current, "0") == 0) {
        if(ch == '.') {
            set_current(e, "0.");
        } else {
            set_current(e, (char[]){ch, '\0'});
        }
        e->should_reset = false;
    } else {
        size_t len = strlen(e->current);
        if(len < sizeof(e->current) - 2) {
            e->current[len] = ch;
            e->current[len + 1] = '\0';
        }
    }
    e->is_result = false;
}

static void calculate(calc_engine_t * e) {
    if(e->op == 0 || e->previous[0] == '\0') return;
    double prev = strtod(e->previous, NULL);
    double curr = strtod(e->current, NULL);
    if(e->op == '÷' && curr == 0) {
        set_current(e, "Error");
        e->op = 0;
        e->previous[0] = '\0';
        e->should_reset = true;
        e->expression[0] = '\0';
        return;
    }
    finish_result(e, compute(prev, curr, e->op));
}

static void handle_operator(calc_engine_t * e, char op) {
    if(is_error(e)) {
        /* Error 态运算符先清除 */
        calc_engine_init(e);
    }
    if(e->op != 0 && !e->should_reset) {
        calculate(e);
    }
    snprintf(e->previous, sizeof(e->previous), "%s", e->current);
    e->op = op;
    update_expression_op(e);
    e->should_reset = true;
}

static void handle_advanced(calc_engine_t * e, calc_btn_t btn) {
    if(is_error(e) && btn != CALC_BTN_LPAREN && btn != CALC_BTN_RPAREN) {
        calc_engine_init(e);
    }
    double num = strtod(e->current, NULL);
    double result = 0;
    bool has_result = true;
    switch(btn) {
        case CALC_BTN_SIN: result = sin(num); break;
        case CALC_BTN_COS: result = cos(num); break;
        case CALC_BTN_TAN: result = tan(num); break;
        case CALC_BTN_LOG: result = log10(num); break;
        case CALC_BTN_LN:  result = log(num); break;
        case CALC_BTN_SQRT:
            if(num < 0) { set_current(e, "Error"); e->should_reset = true; return; }
            result = sqrt(num); break;
        case CALC_BTN_SQUARE: result = pow(num, 2); break;
        case CALC_BTN_CUBE:   result = pow(num, 3); break;
        case CALC_BTN_FACTORIAL:
            if(num < 0 || floor(num) != num) {
                set_current(e, "Error"); e->should_reset = true; return;
            }
            result = 1;
            for(int i = 2; i <= (int)num; i++) result *= i;
            break;
        case CALC_BTN_LPAREN: {
            /* 与 JS 一致：追加 '(' 到 current */
            size_t len = strlen(e->current);
            if(len < sizeof(e->current) - 2 && !(len == 1 && e->current[0] == '0')) {
                e->current[len] = '('; e->current[len+1] = '\0';
            } else if(len == 1 && e->current[0] == '0') {
                set_current(e, "(");
            }
            return;
        }
        case CALC_BTN_RPAREN: {
            size_t len = strlen(e->current);
            if(len < sizeof(e->current) - 2) {
                e->current[len] = ')'; e->current[len+1] = '\0';
            }
            return;
        }
        default: has_result = false; break;
    }
    if(has_result) {
        finish_result(e, result);
    }
}

const char * calc_engine_press(calc_engine_t * e, calc_btn_t btn) {
    switch(btn) {
        case CALC_BTN_0: case CALC_BTN_1: case CALC_BTN_2: case CALC_BTN_3:
        case CALC_BTN_4: case CALC_BTN_5: case CALC_BTN_6: case CALC_BTN_7:
        case CALC_BTN_8: case CALC_BTN_9:
            handle_number(e, (char)('0' + (btn - CALC_BTN_0)));
            break;
        case CALC_BTN_DOT:
            if(is_error(e)) { set_current(e, "0"); }
            if(!strchr(e->current, '.')) {
                if(e->should_reset) {
                    set_current(e, "0.");
                    e->should_reset = false;
                } else {
                    size_t len = strlen(e->current);
                    if(len < sizeof(e->current) - 2) {
                        e->current[len] = '.'; e->current[len+1] = '\0';
                    }
                }
                e->is_result = false;
            }
            break;
        case CALC_BTN_ADD: handle_operator(e, '+'); break;
        case CALC_BTN_SUB: handle_operator(e, '-'); break;
        case CALC_BTN_MUL: handle_operator(e, '×'); break;
        case CALC_BTN_DIV: handle_operator(e, '÷'); break;
        case CALC_BTN_POW: handle_operator(e, '^'); break;
        case CALC_BTN_EQ:  calculate(e); break;
        case CALC_BTN_CLEAR: calc_engine_init(e); break;
        case CALC_BTN_DELETE: {
            if(is_error(e)) { calc_engine_init(e); break; }
            size_t len = strlen(e->current);
            if(len > 1) {
                e->current[len - 1] = '\0';
            } else {
                set_current(e, "0");
            }
            break;
        }
        case CALC_BTN_SIGN: {
            if(is_error(e) || strcmp(e->current, "0") == 0) break;
            if(e->current[0] == '-') {
                memmove(e->current, e->current + 1, strlen(e->current));
            } else {
                size_t len = strlen(e->current);
                if(len < sizeof(e->current) - 1) {
                    memmove(e->current + 1, e->current, len + 1);
                    e->current[0] = '-';
                }
            }
            break;
        }
        case CALC_BTN_PERCENT: {
            if(is_error(e)) break;
            double num = strtod(e->current, NULL) / 100.0;
            snprintf(e->current, sizeof(e->current), "%.12g", num);
            e->should_reset = true;
            break;
        }
        default:
            /* 科学键 */
            if(btn >= CALC_BTN_SIN && btn <= CALC_BTN_FACTORIAL) {
                handle_advanced(e, btn);
            }
            break;
    }
    return e->current;
}
```

`touching_cb` 里设位移拖拽分支（`icon_shake` 或 `lv_obj_set_pos` 移动时）置 `did_drag = 1`。`touch_time_count` 进入抖动分支（>70）时也置 `did_drag = 1`（长按抖动不算短按）。

`released_cb` 开头重置前记录：在 `old_index = meta->index;` 后加对 `did_drag` 的判断不在这里——短按判定放 released_cb **末尾**，所有拖拽换位分支都提前 `return`，所以末尾到达 = 未拖拽。

released_cb 末尾（函数结束前）加：
```c
if(!did_drag && !icon_shake) {
    /* 短按启动 app：查 desktop_apps 找当前 meta 对应项 */
    int app_idx = -1;
    int seen = 0;
    for(int a = 0; a < (int)desktop_app_count; a++) {
        if(desktop_apps[a].page == meta->page) {
            if(seen == meta->index) { app_idx = a; break; }
            seen++;
        }
    }
    if(app_idx >= 0 && desktop_apps[app_idx].launch != NULL) {
        desktop_apps[app_idx].launch();
    }
}
did_drag = 0;
```

> ⚠️ 现有 `page_icon_count_init` page_0=0,page_1=10,page_2=2，与 `desktop_apps[].page` 字段对应。`seen` 计数按 page 内顺序匹配 meta->index。需实测确认 index 语义与填充顺序一致（desktop.c 填充循环用 `slot_k[pg_idx]` 递增，meta->index = 槽位 k，一致）。
